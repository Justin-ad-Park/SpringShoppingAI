// JSON 라이브러리로 생성하는 코드의 발췌이다. 이 평가에는 빌드·실행 환경을 제공하지 않는다.
// JSON은 프로젝트에서 준비한 ObjectMapper이며, ObjectNode는 해당 라이브러리의 타입이다.
static ObjectNode createMessageResponseBody(String model, String answer) {
    var body = JSON.createObjectNode().put("type", "message").put("role", "assistant").put("model", model);
    body.putArray("content").addObject().put("type", "text").put("text", answer);
    body.putObject("usage").put("input_tokens", 10).put("output_tokens", 5);
    return body;
}
